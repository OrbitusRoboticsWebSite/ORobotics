REM RedAgent 0.4.0 pilot installation - windows
REM Use on your unlocked corporate computer. Match the Flipper keyboard layout to the host.
REM Linux requires Ctrl+Alt+T to open a terminal. macOS requires Cmd+Space for Spotlight.
REM Leaves the terminal visible. Installs for this user; no elevation, tests, or MDM enrollment.
DEFAULT_STRING_DELAY 10
DELAY 2000
GUI r
DELAY 1200
STRING powershell.exe -NoLogo -NoProfile
ENTER
DELAY 4000
STRING & { $ErrorActionPreference='Stop'; if ($ExecutionContext.SessionState.LanguageMode -ne 'FullLanguage') { throw 'Use your administrator-approved installer: PowerShell language is restricted.' }; if ((Get-ExecutionPolicy -Scope MachinePolicy) -ne 'Undefined' -or (Get-ExecutionPolicy -Scope UserPolicy) -ne 'Undefined' -or (Get-ExecutionPolicy) -eq 'AllSigned') { throw 'Use your administrator-approved installer: a managed or signing policy is configured.' }; $p=Join-Path $env:TEMP ('redagent-'+[guid]::NewGuid()+'.ps1'); try { Invoke-WebRequest -UseBasicParsing https://www.orbitusrobotics.com/r/install.ps1 -OutFile $p -TimeoutSec 120 -ErrorAction Stop; $b=[IO.File]::ReadAllBytes($p); $h=[Security.Cryptography.SHA256]::Create(); try { $s=[BitConverter]::ToString($h.ComputeHash($b)).Replace('-','').ToLowerInvariant() } finally { $h.Dispose() }; if ($s -ne 'dedb827133591d81619802bba816a6ae5a50ad90d6fc2a03343a6a2b53b364af') { throw 'Installer SHA-256 mismatch. Download the latest command or Flipper script from the website.' }; Write-Host ('Verified installer: '+$p); & ([scriptblock]::Create([Text.Encoding]::UTF8.GetString($b))) } finally { if (Test-Path -LiteralPath $p) { Remove-Item -LiteralPath $p -Force } } }
ENTER
